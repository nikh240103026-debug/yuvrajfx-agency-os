import { existsSync } from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

export const ROOT = path.resolve(
  path.dirname(fileURLToPath(import.meta.url)),
  '../..',
);

export type SmtpConfig = {
  host: string;
  port: number;
  secure: boolean;
  user: string | null;
  pass: string | null;
  from: string;
};

export type Config = {
  host: string;
  port: number;
  dbPath: string;
  databaseUrl: string | null;
  cookieSecure: boolean;
  isDev: boolean;
  smtp: SmtpConfig | null;
};

function bool(v: string | undefined, fallback = false): boolean {
  if (v === undefined || v === '') return fallback;
  return ['1', 'true', 'yes', 'on'].includes(v.trim().toLowerCase());
}

function flag(argv: string[], name: string): string | undefined {
  const hit = argv.find((a) => a.startsWith(`--${name}=`));
  return hit ? hit.slice(name.length + 3) : undefined;
}

export function loadConfig(argv: string[] = process.argv): Config {
  const envFile = path.join(ROOT, '.env');

  if (existsSync(envFile)) {
    process.loadEnvFile(envFile);
  }

  const env = process.env;
  const dbPath =
    flag(argv, 'db') || env.DB_PATH?.trim() || 'data/agency.db';

  const smtpHost = env.SMTP_HOST?.trim();
  const databaseUrl = env.DATABASE_URL?.trim() || null;

  return {
    host: env.HOST?.trim() || '127.0.0.1',
    port: Number(flag(argv, 'port')) || Number(env.PORT) || 4747,
    dbPath:
      dbPath === ':memory:'
        ? dbPath
        : path.resolve(ROOT, dbPath),
    databaseUrl,
    cookieSecure: bool(env.COOKIE_SECURE),
    isDev: argv.includes('--dev'),
    smtp: smtpHost
      ? {
          host: smtpHost,
          port: Number(env.SMTP_PORT) || 587,
          secure: bool(env.SMTP_SECURE),
          user: env.SMTP_USER?.trim() || null,
          pass: env.SMTP_PASS || null,
          from:
            env.SMTP_FROM?.trim() ||
            env.SMTP_USER?.trim() ||
            'no-reply@localhost',
        }
      : null,
  };
}

-- Database-level guarantees behind "never mark an invoice paid unless payments cover it":
-- payments can't exceed the balance, can't change currency, and a total can't drop below what was paid.
CREATE VIEW invoice_balances AS
SELECT i.*,
       COALESCE(p.paid_minor, 0)                 AS paid_minor,
       i.total_minor - COALESCE(p.paid_minor, 0) AS balance_minor,
       COALESCE(p.payment_count, 0)              AS payment_count,
       p.last_paid_at                            AS last_paid_at
FROM invoices i
LEFT JOIN (
  SELECT invoice_id, SUM(amount_minor) AS paid_minor, COUNT(*) AS payment_count, MAX(paid_at) AS last_paid_at
  FROM payments GROUP BY invoice_id
) p ON p.invoice_id = i.id;

CREATE OR REPLACE FUNCTION yuvrajfx_payments_guard_fn() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE inv_state TEXT; inv_currency TEXT; inv_client BIGINT; inv_total BIGINT; already_paid BIGINT;
BEGIN
  SELECT state, currency, client_id, total_minor INTO inv_state, inv_currency, inv_client, inv_total
  FROM invoices WHERE id = NEW.invoice_id FOR UPDATE;
  IF inv_state IS NULL THEN RAISE EXCEPTION 'PAYMENT_INVOICE_NOT_FOUND'; END IF;
  IF inv_state = 'void' THEN RAISE EXCEPTION 'PAYMENT_INVOICE_VOID'; END IF;
  IF NEW.currency <> inv_currency THEN RAISE EXCEPTION 'PAYMENT_CURRENCY_MISMATCH'; END IF;
  IF NEW.client_id <> inv_client THEN RAISE EXCEPTION 'PAYMENT_CLIENT_MISMATCH'; END IF;
  SELECT COALESCE(SUM(amount_minor), 0) INTO already_paid FROM payments WHERE invoice_id = NEW.invoice_id;
  IF NEW.amount_minor > inv_total - already_paid THEN RAISE EXCEPTION 'PAYMENT_EXCEEDS_BALANCE'; END IF;
  RETURN NEW;
END $$;
CREATE TRIGGER payments_guard BEFORE INSERT ON payments FOR EACH ROW EXECUTE FUNCTION yuvrajfx_payments_guard_fn();

CREATE OR REPLACE FUNCTION yuvrajfx_payments_immutable_fn() RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
  IF NEW.invoice_id IS DISTINCT FROM OLD.invoice_id OR NEW.client_id IS DISTINCT FROM OLD.client_id
     OR NEW.amount_minor IS DISTINCT FROM OLD.amount_minor OR NEW.currency IS DISTINCT FROM OLD.currency THEN
    RAISE EXCEPTION 'PAYMENT_IMMUTABLE';
  END IF;
  RETURN NEW;
END $$;
CREATE TRIGGER payments_immutable BEFORE UPDATE ON payments FOR EACH ROW EXECUTE FUNCTION yuvrajfx_payments_immutable_fn();

CREATE OR REPLACE FUNCTION yuvrajfx_invoice_total_covers_payments_fn() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE already_paid BIGINT;
BEGIN
  SELECT COALESCE(SUM(amount_minor), 0) INTO already_paid FROM payments WHERE invoice_id = NEW.id;

  RETURN NEW;
END $$;

export function saveAutomation(ctx: AppContext, key: AutomationKey, patch: { enabled?: boolean; config?: Record<string, unknown> }): void {

     ON CONFLICT(key) DO UPDATE SET enabled = excluded.enabled, config = excluded.config`,
    { key, enabled: enabled ? 1 : 0, config: JSON.stringify(config) },
  );
}

/**
 * Draft an invoice email (cover note or payment reminder) into the outbox. Returns the outbox id,
 * or null when the client has no email address. Never sends anything.
 */
export function draftInvoiceEmail(ctx: AppContext, invoiceId: number, purpose: 'send' | 'reminder', trigger: 'manual' | 'automation', dedupeKey?: string): number | null {
  const inv = getInvoice(ctx, invoiceId);
  if (!inv) throw notFound('Invoice');
  if (inv.state === 'void') throw conflict('This invoice is void.');
  const client = inv.client as { name: string; company: string | null; email: string | null } | undefined;
  if (!client?.email) return null;
  const s = ctx.settings();
  const amount = formatMoney(inv.total_minor, inv.currency, { decimals: 'always' });
  const balance = formatMoney(inv.balance_minor, inv.currency, { decimals: 'always' });
  const lines =
    purpose === 'reminder'
      ? [
          `Hi ${firstName(client.name)},`,
          '',
          `A friendly reminder that invoice ${inv.number} (${amount}) was due on ${formatDateLong(inv.due_date)}${inv.days_overdue > 0 ? ` â€” ${inv.days_overdue} day${inv.days_overdue === 1 ? '' : 's'} ago` : ''}.`,
          inv.paid_minor > 0 ? `Thank you for the payment received so far; the remaining balance is ${balance}.` : `The outstanding balance is ${balance}.`,
          '',
          ...(s.payment_instructions ? ['Payment details:', s.payment_instructions, ''] : []),
          'If the payment is already on its way, please ignore this note â€” and thank you!',
          '',
          'Best regards,',
          s.business_name,
        ]
      : [
          `Hi ${firstName(client.name)},`,
          '',
          `Please find the details of invoice ${inv.number} below.`,
          '',
          `Amount due: ${balance}`,
          `Issued: ${formatDateLong(inv.issue_date)}`,
          `Due by: ${formatDateLong(inv.due_date)}`,
          '',
          ...(s.payment_instructions ? ['Payment details:', s.payment_instructions, ''] : []),
          'Let me know if you have any questions.',
          '',
          'Best regards,',
          s.business_name,
        ];
  const now = ctx.nowIso();
  const r = ctx.db.run(
    `INSERT INTO outbox (automation_key, channel, to_address, to_name, subject, body, status, entity_type, entity_id, client_id, dedupe_key, is_demo, created_at, updated_at)
     VALUES (:key, 'email', :to, :to_name, :subject, :body, 'pending_approval', 'invoice', :invoice_id, :client_id, :dedupe, :is_demo, :now, :now)
     ON CONFLICT(dedupe_key) DO NOTHING`,
    {
      key: trigger === 'automation' ? 'overdue_invoice_reminders' : null,
      to: client.email,
      to_name: client.name,
      subject: purpose === 'reminder' ? `Payment reminder: invoice ${inv.number} (${balance} outstanding)` : `Invoice ${inv.number} from ${s.business_name}`,
      body: lines.join('\n'),
      invoice_id: inv.id,
      client_id: inv.client_id,
      dedupe: dedupeKey ?? null,
      is_demo: inv.is_demo,
      now,
    },
  );
  if (r.changes === 0) return null;
  return r.lastInsertRowid;
}

function runDeadlineReminders(ctx: AppContext, cfg: Record<string, unknown>) {
  const today = ctx.today();
  const days = Math.max(0, Math.min(30, Number(cfg.days_before) || 2));
  const horizon = addDays(today, days);
  let soon = 0;
  let overdue = 0;
  const due = ctx.db.all<{ id: number; title: string; deadline: string; project_id: number; project_name: string; client_name: string; is_demo: number }>(
    `SELECT d.id, d.title, d.deadline, d.project_id, p.name AS project_name, COALESCE(c.company, c.name) AS client_name, d.is_demo
       FROM deliverables d JOIN projects p ON p.id = d.project_id JOIN clients c ON c.id = p.client_id
      WHERE d.status <> 'delivered' AND d.deadline IS NOT NULL AND d.deadline <= :horizon
        AND p.status NOT IN ('cancelled','on_hold') AND p.archived_at IS NULL`,
    { horizon },
  );
  for (const d of due) {
    const late = d.deadline < today;
    const created = notify(ctx, {
      kind: late ? 'deliverable_overdue' : 'deadline_soon',
      audience: 'staff',
      severity: late ? 'critical' : 'warning',
      title: late ? `Overdue: â€œ${d.title}â€` : `Due ${diffDays(today, d.deadline) === 0 ? 'today' : diffDays(today, d.deadline) === 1 ? 'tomorrow' : formatDate(d.deadline, today)}: â€œ${d.title}â€`,
      body: `${d.client_name} Â· ${d.project_name}${late ? ` Â· was due ${formatDate(d.deadline, today)}` : ''}`,
      link: `/projects/${d.project_id}`,
      entity_type: 'deliverable',
      entity_id: d.id,
      dedupe_key: `${late ? 'deliverable_overdue' : 'deadline_soon'}:${d.id}:${d.deadline}`,
      is_demo: d.is_demo,
    });
    if (created) late ? overdue++ : soon++;
  }
  const tasks = ctx.db.all<{ id: number; title: string; due_date: string; is_demo: number }>(
    `SELECT id, title, due_date, is_demo FROM tasks WHERE status IN ('todo','in_progress') AND due_date < :today`,
    { today },
  );
  let tasksOverdue = 0;
  for (const t of tasks) {
    if (
      notify(ctx, {
        kind: 'task_overdue',
        audience: 'staff',
        severity: 'warning',
        title: `Task overdue: ${t.title}`,
        body: `Was due ${formatDate(t.due_date, today)}`,
        link: '/tasks?view=overdue',
        entity_type: 'task',
        entity_id: t.id,
        dedupe_key: `task_overdue:${t.id}:${t.due_date}`,
        is_demo: t.is_demo,
      })
    )
      tasksOverdue++;
  }
  return { due_soon: soon, deliverables_overdue: overdue, tasks_overdue: tasksOverdue };
}

function runOverdueInvoices(ctx: AppContext, cfg: Record<string, unknown>) {
  const today = ctx.today();
  const grace = Math.max(0, Number(cfg.grace_days) || 0);
  const every = Math.max(1, Number(cfg.repeat_every_days) || 7);
  const rows = ctx.db.all<{ id: number; number: string; due_date: string; balance_minor: number; currency: string; client_name: string; client_id: number; is_demo: number }>(
    `SELECT b.id, b.number, b.due_date, b.balance_minor, b.currency, COALESCE(c.company, c.name) AS client_name, b.client_id, b.is_demo
       FROM invoice_balances b JOIN clients c ON c.id = b.client_id
      WHERE ${invoiceStatusSql('b')} = 'overdue'`,
    { today },
  );
  let flagged = 0;
  let drafted = 0;
  for (const inv of rows) {
    const late = diffDays(inv.due_date, today);
    if (late < grace) continue;
    const bucket = Math.floor(late / every);
    if (
      notify(ctx, {
        kind: 'invoice_overdue',
        audience: 'admin',
        severity: 'critical',
        title: `Invoice ${inv.number} is ${late} day${late === 1 ? '' : 's'} overdue`,
        body: `${inv.client_name} Â· ${formatMoney(inv.balance_minor, inv.currency, { decimals: 'always' })} outstanding`,
        link: `/invoices/${inv.id}`,
        entity_type: 'invoice',
        entity_id: inv.id,
        dedupe_key: `invoice_overdue:${inv.id}:${bucket}`,
        is_demo: inv.is_demo,
      })
    )
      flagged++;
    if (cfg.draft_client_emails !== false) {
      const already = ctx.db.get(`SELECT 1 FROM outbox WHERE entity_type = 'invoice' AND entity_id = :id AND status = 'pending_approval'`, { id: inv.id });
      if (!already && draftInvoiceEmail(ctx, inv.id, 'reminder', 'automation', `invoice_reminder:${inv.id}:${bucket}`)) drafted++;
    }
  }
  return { overdue_flagged: flagged, reminders_drafted: drafted };
}

function runRenewals(ctx: AppContext, cfg: Record<string, unknown>) {
  const today = ctx.today();
  const days = Math.max(1, Math.min(90, Number(cfg.days_before) || 14));
  const rows = ctx.db.all<{ id: number; name: string; company: string | null; renewal_date: string; is_demo: number }>(
    `SELECT id, name, company, renewal_date, is_demo FROM clients
      WHERE archived_at IS NULL AND status IN ('active','onboarding','paused') AND renewal_date BETWEEN :today AND :until`,
    { today, until: addDays(today, days) },
  );
  let notified = 0;
  let tasks = 0;
  for (const c of rows) {
    const label = c.company || c.name;
    if (
      notify(ctx, {
        kind: 'renewal_soon',
        audience: 'admin',
        severity: 'info',
        title: `${label} renews ${formatDate(c.renewal_date, today)}`,
        body: `Contract renewal in ${diffDays(today, c.renewal_date)} days â€” confirm scope and pricing.`,
        link: `/clients/${c.id}`,
        entity_type: 'client',
        entity_id: c.id,
        dedupe_key: `renewal:${c.id}:${c.renewal_date}`,
        is_demo: c.is_demo,
      })
    ) {
      notified++;
      if (cfg.create_task !== false) {
        const key = `renewal:${c.id}:${c.renewal_date}`;
        if (!ctx.db.get('SELECT 1 FROM tasks WHERE recurrence_key = :key', { key })) {
          ctx.db.insert('tasks', {
            title: `Renewal check-in with ${label}`,
            description: `Contract renews on ${formatDateLong(c.renewal_date)}. Review results, scope and pricing before then.`,
            client_id: c.id,
            priority: 'high',
            due_date: addDays(c.renewal_date, -3) < today ? today : addDays(c.renewal_date, -3),
            status: 'todo',
            recurrence_key: key,
            source: 'automation',
            is_demo: c.is_demo,
            created_at: ctx.nowIso(),
            updated_at: ctx.nowIso(),
          });
          tasks++;
        }
      }
    }
  }
  return { renewals_notified: notified, tasks_created: tasks };
}

function runFollowUps(ctx: AppContext) {
  const today = ctx.today();
  const rows = ctx.db.all<{ id: number; name: string; company: string | null; next_follow_up: string; stage: string; is_demo: number }>(
    `SELECT id, name, company, next_follow_up, stage, is_demo FROM leads WHERE stage NOT IN ('won','lost') AND next_follow_up <= :today`,
    { today },
  );
  let n = 0;
  for (const l of rows) {
    const late = l.next_follow_up < today;
    if (
      notify(ctx, {
        kind: 'follow_up_due',
        audience: 'admin',
        severity: late ? 'warning' : 'info',
        title: `Follow up with ${l.company ? `${l.name} (${l.company})` : l.name}`,
        body: late ? `Follow-up was due ${formatDate(l.next_follow_up, today)}` : 'Follow-up is due today',
        link: `/pipeline?lead=${l.id}`,
        entity_type: 'lead',
        entity_id: l.id,
        dedupe_key: `follow_up:${l.id}:${l.next_follow_up}`,
        is_demo: l.is_demo,
      })
    )
      n++;
  }
  return { follow_ups_notified: n };
}

let running = false;

export function runAutomations(ctx: AppContext, trigger: 'startup' | 'schedule' | 'manual', only?: AutomationKey[]): { run_id: number; summary: RunSummary; status: 'ok' | 'error' } {
  if (running) throw conflict('Automations are already running. Try again in a moment.');
  running = true;
  const started = ctx.nowIso();
  const runId = ctx.db.insert('automation_runs', { trigger, started_at: started, status: 'running', summary: {} });
  const summary: RunSummary = {};
  let status: 'ok' | 'error' = 'ok';
  let firstError: string | null = null;
  try {
    // Only run when the agency exists (setup done), so an empty install stays quiet.
    if (!ctx.db.get('SELECT 1 FROM users LIMIT 1')) {
      summary.skipped = 'Setup not completed yet.';
    } else {
      for (const key of AUTOMATION_KEYS) {
        if (only && !only.includes(key)) continue;
        const { enabled, config } = automationConfig(ctx, key);
        if (!enabled) {
          summary[key] = 'disabled';
          continue;
        }
        try {
          let result: Record<string, number>;
          switch (key) {
            case 'deadline_reminders':
              result = runDeadlineReminders(ctx, config);
              break;
            case 'overdue_invoice_reminders':
              result = runOverdueInvoices(ctx, config);
              break;
            case 'renewal_reminders':
              result = runRenewals(ctx, config);
              break;
            case 'follow_up_reminders':
              result = runFollowUps(ctx);
              break;
            case 'recurring_work':
              result = materializeRecurring(ctx, SYSTEM_ACTOR);
              break;
            case 'monthly_revenue_summary':
              result = runMonthly(ctx, config);
              break;
            case 'weekly_report':
              result = runWeekly(ctx, config);
              break;
          }
          summary[key] = result;
          ctx.db.run(
            `INSERT INTO automation_settings (key, enabled, config, last_run_at, last_result) VALUES (:key, 1, '{}', :at, :result)
             ON CONFLICT(key) DO UPDATE SET last_run_at = excluded.last_run_at, last_result = excluded.last_result`,
            { key, at: ctx.nowIso(), result: JSON.stringify(result) },
          );
        } catch (err) {
          status = 'error';
          const msg = err instanceof Error ? err.message : String(err);
          firstError ??= `${key}: ${msg}`;
          summary[key] = `error: ${msg}`;
          console.error(`[automation] ${key} failed`, err);
        }
      }
    }
  } finally {
    ctx.db.run('UPDATE automation_runs SET finished_at = :at, status = :status, summary = :summary, error = :error WHERE id = :id', {
      at: ctx.nowIso(),
      status,
      summary: JSON.stringify(summary),
      error: firstError,
      id: runId,
    });
    running = false;
  }
  return { run_id: runId, summary, status };
}

/** Runs automations shortly after start-up and then every 15 minutes while the app is running. */
export function startScheduler(ctx: AppContext, everyMs = 15 * 60_000): () => void {
  const tick = (trigger: 'startup' | 'schedule') => {
    try {
      runAutomations(ctx, trigger);
    } catch (err) {
      if (!(err instanceof Error && err.message.includes('already running'))) console.error('[automation] run failed', err);
    }
  };
  const first = setTimeout(() => tick('startup'), 3000);
  const timer = setInterval(() => tick('schedule'), everyMs);
  first.unref?.();
  timer.unref?.();
  return () => {
    clearTimeout(first);
    clearInterval(timer);
  };
}

type OutboxRow = { id: number; status: string; to_address: string; to_name: string | null; subject: string; body: string; entity_type: string | null; entity_id: number | null; client_id: number | null; is_demo: number };

export async function approveAndSend(ctx: AppContext, actor: Actor, id: number): Promise<{ status: string; error?: string }> {
  const row = ctx.db.get<OutboxRow>('SELECT * FROM outbox WHERE id = :id', { id });
  if (!row) throw notFound('Message');
  if (row.status !== 'pending_approval' && row.status !== 'failed') throw conflict('This message was already handled.');
  if (!ctx.mailer.configured) throw conflict('Email is not set up (no SMTP settings). Copy the message into your own email, then use â€œMark as sent manuallyâ€.');
  const now = ctx.nowIso();
  ctx.db.run(`UPDATE outbox SET approved_by = :uid, approved_at = :now, updated_at = :now WHERE id = :id`, { uid: actor.id, now, id });
  try {
    await ctx.mailer.send({ to: row.to_address, toName: row.to_name, subject: row.subject, text: row.body });
    ctx.db.run(`UPDATE outbox SET status = 'sent', sent_at = :now, error = NULL, updated_at = :now WHERE id = :id`, { now: ctx.nowIso(), id });
    logActivity(ctx, actor, { entity_type: row.entity_type ?? 'outbox', entity_id: row.entity_id ?? id, client_id: row.client_id, action: 'email_sent', summary: `Email sent to ${row.to_address}: ${row.subject}`, is_demo: row.is_demo });
    if (row.entity_type === 'invoice' && row.entity_id && row.subject.startsWith('Invoice ')) {
      const inv = ctx.db.get<{ state: string }>('SELECT state FROM invoices WHERE id = :id', { id: row.entity_id });
      if (inv?.state === 'draft') ctx.db.run(`UPDATE invoices SET state = 'sent', sent_at = :now, updated_at = :now WHERE id = :id`, { now: ctx.nowIso(), id: row.entity_id });
    }
    return { status: 'sent' };
  } catch (err) {
    const msg = err instanceof Error ? err.message : String(err);
    ctx.db.run(`UPDATE outbox SET status = 'failed', error = :err, updated_at = :now WHERE id = :id`, { err: msg.slice(0, 1000), now: ctx.nowIso(), id });
    return { status: 'failed', error: msg };
  }
}

export function markSentManually(ctx: AppContext, actor: Actor, id: number): void {
  const row = ctx.db.get<OutboxRow>('SELECT * FROM outbox WHERE id = :id', { id });
  if (!row) throw notFound('Message');
  if (row.status === 'sent' || row.status === 'sent_manually') return;
  const now = ctx.nowIso();
  ctx.db.run(`UPDATE outbox SET status = 'sent_manually', approved_by = :uid, approved_at = :now, sent_at = :now, updated_at = :now WHERE id = :id`, { uid: actor.id, now, id });
  logActivity(ctx, actor, { entity_type: row.entity_type ?? 'outbox', entity_id: row.entity_id ?? id, client_id: row.client_id, action: 'email_sent_manually', summary: `Marked as sent manually to ${row.to_address}: ${row.subject}`, is_demo: row.is_demo });
}
