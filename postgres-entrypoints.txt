/** The API. Static files / the Vite dev server are mounted by index.ts after this. */
export function createApp(ctx: AppContext): Express {
  const app = express();
  app.disable('x-powered-by');
  app.set('trust proxy', false);
  app.set('query parser', 'simple');

  app.use(
    helmet({
      contentSecurityPolicy: ctx.config.isDev
        ? false
        : {
            directives: {
              defaultSrc: ["'self'"],
              scriptSrc: ["'self'"],
              styleSrc: ["'self'", "'unsafe-inline'"],
              imgSrc: ["'self'", 'data:', 'blob:'],
              fontSrc: ["'self'", 'data:'],
              connectSrc: ["'self'"],
              frameAncestors: ["'none'"],
              formAction: ["'self'"],
              objectSrc: ["'none'"],
              baseUri: ["'self'"],
              upgradeInsecureRequests: null,
            },
          },
      crossOriginEmbedderPolicy: false,
      hsts: ctx.config.cookieSecure,
    }),
  );

  app.use('/api', express.json({ limit: '5mb' }));
  app.use('/api', (_req, res, next) => {
    res.setHeader('Cache-Control', 'no-store');
    next();
  });
  app.use('/api', sessionMiddleware(ctx));
  app.use('/api', csrfGuard);

  app.get('/api/health', (_req, res) => {
    res.json({ ok: true, time: ctx.nowIso() });
  });

  app.use('/api/auth', authRouter(ctx));
  app.use('/api/portal', requireClient, portalRouter(ctx));

  // Staff (owner, admin, team member)
  app.use('/api/dashboard', requireStaff, dashboardRouter(ctx));
  app.use('/api/clients', requireStaff, clientsRouter(ctx));
  app.use('/api/team', requireStaff, teamRouter(ctx));
  app.use('/api/projects', requireStaff, projectsRouter(ctx));
  app.use('/api/deliverables', requireStaff, deliverablesRouter(ctx));
  app.use('/api/feedback', requireStaff, feedbackRouter(ctx));
  app.use('/api/tasks', requireStaff, tasksRouter(ctx));
  app.use('/api/recurring', requireStaff, recurringRouter(ctx));
  app.use('/api/calendar', requireStaff, calendarRouter(ctx));
  app.use('/api/notifications', requireStaff, notificationsRouter(ctx));
  app.use('/api/activity', requireStaff, activityRouter(ctx));
  app.use('/api/search', requireStaff, searchRouter(ctx));

  // Owner/admin only: money, sales, automation, settings
  app.use('/api/invoices', requireAdmin, invoicesRouter(ctx));
  app.use('/api/payments', requireAdmin, paymentsRouter(ctx));
  app.use('/api/expenses', requireAdmin, expensesRouter(ctx));
  app.use('/api/finance', requireAdmin, financeRouter(ctx));
  app.use('/api/leads', requireAdmin, leadsRouter(ctx));
  app.use('/api/automations', requireAdmin, automationsRouter(ctx));
  app.use('/api/outbox', requireAdmin, outboxRouter(ctx));
  app.use('/api/settings', requireAdmin, settingsRouter(ctx));

  app.use('/api', (_req, _res, next) => next(new HttpError(404, 'Unknown API endpoint.')));
  app.use('/api', errorHandler);
  return app;
}

export type AppContext = {
  db: Db;
  config: Config;
  clock: Clock;
  mailer: Mailer;
  nowIso(): string;
  /** Today's date in the business timezone (Settings â†’ timezone). */
  today(): string;
  settings(): Settings;
  invalidateSettings(): void;
};

export function createContext(opts: { db: Db; config: Config; mailer: Mailer; clock?: Clock }): AppContext {
  const clock = opts.clock ?? { now: () => new Date() };
  let cached: Settings | null = null;
  const ctx: AppContext = {
    db: opts.db,
    config: opts.config,
    clock,
    mailer: opts.mailer,
    nowIso: () => clock.now().toISOString(),
    today: () => todayIn(ctx.settings().timezone, clock.now()),
    settings: () => (cached ??= readSettings(opts.db)),
    invalidateSettings: () => {
      cached = null;
    },
  };
  return ctx;
}

import { DatabaseSync, type StatementSync } from 'node:sqlite';

export type Row = Record<string, unknown>;
export type Params = Record<string, unknown> | readonly unknown[] | undefined;
type Bindable = null | number | bigint | string | Uint8Array;
type Prepared = { stmt: StatementSync; names: string[] };

const NAME_RE = /[:@$]([A-Za-z_][A-Za-z0-9_]*)/g;
const IDENT_RE = /^[a-z_][a-z0-9_]*$/;

/** Named parameters used by a statement, ignoring anything inside string literals or comments. */
function paramNames(sql: string): string[] {
  const stripped = sql.replace(/'(?:[^']|'')*'/g, "''").replace(/--[^\n]*/g, '');
  const names = new Set<string>();
  for (const m of stripped.matchAll(NAME_RE)) names.add(m[1]);
  return [...names];
}

function bindValue(v: unknown): Bindable {
  if (v === undefined || v === null) return null;
  if (typeof v === 'boolean') return v ? 1 : 0;
  if (typeof v === 'number' || typeof v === 'string' || typeof v === 'bigint') return v;
  if (v instanceof Uint8Array) return v;
  if (v instanceof Date) return v.toISOString();
  return JSON.stringify(v);
}

/**
 * Thin synchronous wrapper over node:sqlite.
 * - named params (:name) bind from an object; unknown keys are ignored, undefined becomes NULL
 * - booleans become 0/1, arrays/objects are stored as JSON text
 * - tx() nests with savepoints, so services can compose
 */
export class Db {
  readonly raw: DatabaseSync;
  #cache = new Map<string, Prepared>();
  #depth = 0;

  constructor(file: string) {
    this.raw = new DatabaseSync(file);
    this.raw.exec('PRAGMA foreign_keys = ON; PRAGMA busy_timeout = 5000;');
    if (file !== ':memory:') this.raw.exec('PRAGMA journal_mode = WAL; PRAGMA synchronous = NORMAL;');
  }

  #prepare(sql: string): Prepared {
    let p = this.#cache.get(sql);
    if (!p) {
      p = { stmt: this.raw.prepare(sql), names: paramNames(sql) };
      if (this.#cache.size > 500) this.#cache.clear();
      this.#cache.set(sql, p);
    }
    return p;
  }

  #bind(p: Prepared, params: Params): Bindable[] | [Record<string, Bindable>] {
    if (params === undefined) {
      // SQLite would silently bind a forgotten :param as NULL; fail loudly instead.
      if (p.names.length) throw new Error(`Missing SQL parameter(s): ${p.names.join(', ')}`);
      return [];
    }
    if (Array.isArray(params)) return params.map(bindValue);
    const src = params as Record<string, unknown>;
    const obj: Record<string, Bindable> = {};
    for (const n of p.names) {
      if (!(n in src)) throw new Error(`Missing SQL parameter: ${n}`);
      obj[n] = bindValue(src[n]);
    }
    return [obj];
  }

  all<T = Row>(sql: string, params?: Params): T[] {
    const p = this.#prepare(sql);
    return p.stmt.all(...(this.#bind(p, params) as [])) as T[];
  }

  get<T = Row>(sql: string, params?: Params): T | undefined {
    const p = this.#prepare(sql);
    return p.stmt.get(...(this.#bind(p, params) as [])) as T | undefined;
  }

  /** First column of the first row. */
  value<T = unknown>(sql: string, params?: Params): T | undefined {
    const row = this.get<Row>(sql, params);
    if (!row) return undefined;
    const k = Object.keys(row)[0];
    return row[k] as T;
  }

  run(sql: string, params?: Params): { changes: number; lastInsertRowid: number } {
    const p = this.#prepare(sql);
    const r = p.stmt.run(...(this.#bind(p, params) as []));
    return { changes: Number(r.changes), lastInsertRowid: Number(r.lastInsertRowid) };
  }

  exec(sql: string): void {
    this.raw.exec(sql);
  }

  /** INSERT a row from an object whose keys are column names; returns the new id. */
  insert(table: string, data: Record<string, unknown>): number {
    const cols = Object.keys(data).filter((k) => data[k] !== undefined);
    for (const c of [table, ...cols]) if (!IDENT_RE.test(c)) throw new Error(`Bad identifier: ${c}`);
    const sql = `INSERT INTO ${table} (${cols.join(', ')}) VALUES (${cols.map((c) => ':' + c).join(', ')})`;
    return this.run(sql, data).lastInsertRowid;
  }

  /** UPDATE the given columns of one row by id; returns the number of changed rows. */
  update(table: string, id: number, data: Record<string, unknown>): number {
    const cols = Object.keys(data).filter((k) => data[k] !== undefined && k !== 'id');
    if (cols.length === 0) return 0;
    for (const c of [table, ...cols]) if (!IDENT_RE.test(c)) throw new Error(`Bad identifier: ${c}`);
    const sql = `UPDATE ${table} SET ${cols.map((c) => `${c} = :${c}`).join(', ')} WHERE id = :row_id`;
    return this.run(sql, { ...data, row_id: id }).changes;
  }

  tx<T>(fn: () => T): T {
    const sp = `sp_${this.#depth}`;
    this.raw.exec(this.#depth === 0 ? 'BEGIN IMMEDIATE' : `SAVEPOINT ${sp}`);
    this.#depth++;
    try {
      const out = fn();
      if (out instanceof Promise) throw new Error('Db.tx callbacks must be synchronous');
      this.#depth--;
      this.raw.exec(this.#depth === 0 ? 'COMMIT' : `RELEASE ${sp}`);
      return out;
    } catch (err) {
      this.#depth--;
      if (this.#depth === 0) this.raw.exec('ROLLBACK');
      else this.raw.exec(`ROLLBACK TO ${sp}; RELEASE ${sp};`);
      throw err;
    }
  }

  close(): void {
    this.#cache.clear();
    this.raw.close();
  }
}

/** Parse a JSON text column, falling back when it is empty or malformed. */
export function json<T>(value: unknown, fallback: T): T {
  if (typeof value !== 'string' || value === '') return fallback;
  try {
    return JSON.parse(value) as T;
  } catch {
    return fallback;
  }
}

import { AsyncLocalStorage } from 'node:async_hooks';
import { Pool, type PoolClient, type QueryResult } from 'pg';

export type Row = Record<string, unknown>;
export type Params = Record<string, unknown> | readonly unknown[] | undefined;

type QueryContext = { client: PoolClient };

const IDENT_RE = /^[a-z_][a-z0-9_]*$/;


function bindValue(value: unknown): unknown {
  if (value === undefined || value === null) return null;
  if (value instanceof Date) return value;
  if (typeof value === 'object' && !(value instanceof Uint8Array)) {
    return JSON.stringify(value);
  }
  return value;
}


/**
 * Converts named SQLite-style parameters (:name, @name, $name)
 * into PostgreSQL positional parameters ($1, $2, ...).
 *
 * Quoted strings, identifiers, comments and PostgreSQL dollar-quoted
 * strings are skipped to avoid replacing text that is not a parameter.
 */
export function compileSql(
  sql: string,
  params?: Params,
): { sql: string; values: unknown[] } {
  if (params === undefined) {
    if (/(?<!:):[a-z_][a-z0-9_]*/i.test(sql)) {
      throw new Error('SQL parameters are missing.');
    }
    return { sql, values: [] };
  }

  if (Array.isArray(params)) {

        // Preserve native PostgreSQL placeholders used by insert() and update().
    // These SQL statements already contain $1, $2, etc.
    if (/\$\d+/.test(sql)) {
      const placeholders = [...sql.matchAll(/\$(\d+)/g)]
        .map((match) => Number(match[1]));

      const maxPlaceholder = Math.max(0, ...placeholders);

      if (maxPlaceholder !== params.length) {
        throw new Error(
          `SQL parameter count mismatch: expected ${maxPlaceholder}, received ${params.length}.`,
        );
      }

      return {
        sql,
        values: params.map(bindValue),
      };
    }

    let index = 0;
    let output = '';
    let state:
      | 'normal'
      | 'single'
      | 'double'
      | 'line-comment'
      | 'block-comment' = 'normal';

    for (let i = 0; i < sql.length; i++) {
      const ch = sql[i];
      const next = sql[i + 1];

      if (state === 'single') {
        output += ch;
        if (ch === "'" && next === "'") output += sql[++i];
        else if (ch === "'") state = 'normal';
        continue;
      }

      if (state === 'double') {
        output += ch;
        if (ch === '"' && next === '"') output += sql[++i];
        else if (ch === '"') state = 'normal';
        continue;
      }

      if (state === 'line-comment') {
        output += ch;
        if (ch === '\n') state = 'normal';
        continue;
      }

      if (state === 'block-comment') {
        output += ch;
        if (ch === '*' && next === '/') {
          output += sql[++i];
          state = 'normal';
        }
        continue;
      }

      if (ch === "'") {
        state = 'single';
        output += ch;
      } else if (ch === '"') {
        state = 'double';
        output += ch;
      } else if (ch === '-' && next === '-') {
        state = 'line-comment';
        output += ch + sql[++i];
      } else if (ch === '/' && next === '*') {
        state = 'block-comment';
        output += ch + sql[++i];
      } else if (ch === '?') {
        output += `$${++index}`;
      } else {
        output += ch;
      }
    }

    if (index !== params.length) {
      throw new Error(
        `SQL parameter count mismatch: expected ${index}, received ${params.length}.`,
      );
    }

    return { sql: output, values: params.map(bindValue) };
  }

  

  const source = params as Record<string, unknown>;
  const names: string[] = [];
  const values: unknown[] = [];
  const indexes = new Map<string, number>();

  let output = '';
  let state:
    | 'normal'
    | 'single'
    | 'double'
    | 'line-comment'
    | 'block-comment' = 'normal';

  for (let i = 0; i < sql.length; i++) {
    const ch = sql[i];
    const next = sql[i + 1];

    if (state === 'single') {
      output += ch;
      if (ch === "'" && next === "'") output += sql[++i];
      else if (ch === "'") state = 'normal';
      continue;
    }

    if (state === 'double') {
      output += ch;
      if (ch === '"' && next === '"') output += sql[++i];
      else if (ch === '"') state = 'normal';
      continue;
    }

    if (state === 'line-comment') {
      output += ch;
      if (ch === '\n') state = 'normal';
      continue;
    }

    if (state === 'block-comment') {
      output += ch;
      if (ch === '*' && next === '/') {
        output += sql[++i];
        state = 'normal';
      }
      continue;
    }

    if (ch === "'") {
      state = 'single';
      output += ch;
      continue;
    }

    if (ch === '"') {
      state = 'double';
      output += ch;
      continue;
    }

    if (ch === '-' && next === '-') {
      state = 'line-comment';
      output += ch + sql[++i];
      continue;
    }

    if (ch === '/' && next === '*') {
      state = 'block-comment';
      output += ch + sql[++i];
      continue;
    }

    if (
      (ch === ':' || ch === '@' || ch === '$') &&
      next &&
      /[a-z_]/i.test(next) &&
      !(ch === ':' && sql[i - 1] === ':')
    ) {
      let end = i + 2;
      while (end < sql.length && /[a-z0-9_]/i.test(sql[end])) end++;

      const name = sql.slice(i + 1, end);

      if (!(name in source)) {
        throw new Error(`Missing SQL parameter: ${name}`);
      }

      let index = indexes.get(name);

      if (index === undefined) {
        index = values.length + 1;
        indexes.set(name, index);
        names.push(name);
        values.push(bindValue(source[name]));
      }

      output += `$${index}`;
      i = end - 1;
      continue;
    }

    output += ch;
  }

  return { sql: output, values };
}

/** Async PostgreSQL adapter. Not yet wired into the application's routes. */
export class PostgresDb {
  readonly pool: Pool;
  #context = new AsyncLocalStorage<QueryContext>();
  #closed = false;

  constructor(connectionString: string) {
    if (!connectionString?.trim()) {
      throw new Error('DATABASE_URL is required for PostgreSQL.');
    }

    this.pool = new Pool({
      connectionString,
      max: Number(process.env.PG_POOL_MAX ?? 10),
      connectionTimeoutMillis: 10_000,
      idleTimeoutMillis: 30_000,
      application_name: 'yuvrajfx-agency-os',
      ssl: process.env.PGSSL === 'true'
        ? { rejectUnauthorized: true }
        : undefined,
    });

    this.pool.on('error', (error) => {
      console.error('Unexpected idle PostgreSQL client error:', error);
    });
  }

  async #query(
    sql: string,
    params?: Params,
  ): Promise<QueryResult<Row>> {
    if (this.#closed) throw new Error('PostgreSQL database is closed.');

    const compiled = compileSql(sql, params);
    const client = this.#context.getStore()?.client;

    if (client) {
      return client.query(compiled.sql, compiled.values) as Promise<QueryResult<Row>>;
    }

    return this.pool.query(compiled.sql, compiled.values) as Promise<QueryResult<Row>>;
  }

  async all<T = Row>(sql: string, params?: Params): Promise<T[]> {
    const result = await this.#query(sql, params);
    return result.rows as T[];
  }

  async get<T = Row>(sql: string, params?: Params): Promise<T | undefined> {
    const result = await this.#query(sql, params);
    return result.rows[0] as T | undefined;
  }

  async value<T = unknown>(sql: string, params?: Params): Promise<T | undefined> {
    const row = await this.get<Row>(sql, params);
    if (!row) return undefined;
    return row[Object.keys(row)[0]] as T;
  }

  async run(
    sql: string,
    params?: Params,
  ): Promise<{ changes: number; lastInsertRowid: number }> {
    const result = await this.#query(sql, params);
    const id = result.rows[0]?.id;

    return {
      changes: result.rowCount ?? 0,
      lastInsertRowid: id === undefined ? 0 : Number(id),
    };
  }

  async exec(sql: string): Promise<void> {
    if (this.#closed) throw new Error('PostgreSQL database is closed.');

    const client = this.#context.getStore()?.client;

    if (client) {
      await client.query(sql);
    } else {
      await this.pool.query(sql);
    }
  }

  async insert(table: string, data: Record<string, unknown>): Promise<number> {
    const columns = Object.keys(data).filter((key) => data[key] !== undefined);

    if (!IDENT_RE.test(table) || columns.length === 0) {
      throw new Error('Invalid table name or empty insert data.');
    }

    for (const column of columns) {
      if (!IDENT_RE.test(column)) throw new Error(`Bad column identifier: ${column}`);
    }

    const values = columns.map((column) => data[column]);
    const placeholders = columns.map((_, index) => `$${index + 1}`);

    const result = await this.#query(
      `INSERT INTO ${table} (${columns.join(', ')}) VALUES (${placeholders.join(', ')}) RETURNING id`,
      values,
    );

    const id = result.rows[0]?.id;

    if (id === undefined) {
      throw new Error(`INSERT into ${table} did not return an id.`);
    }

    return Number(id);
  }

  async update(
    table: string,
    id: number,
    data: Record<string, unknown>,
  ): Promise<number> {
    if (!IDENT_RE.test(table)) throw new Error(`Bad table identifier: ${table}`);

    const columns = Object.keys(data).filter(
      (key) => data[key] !== undefined && key !== 'id',
    );

    if (columns.length === 0) return 0;

    for (const column of columns) {
      if (!IDENT_RE.test(column)) throw new Error(`Bad column identifier: ${column}`);
    }

    const values = columns.map((column) => bindValue(data[column]));
    values.push(id);

    const assignments = columns.map(
      (column, index) => `${column} = $${index + 1}`,
    );

    const result = await this.#query(
      `UPDATE ${table} SET ${assignments.join(', ')} WHERE id = $${values.length}`,
      values,
    );

    return result.rowCount ?? 0;
  }

  async tx<T>(fn: () => Promise<T>): Promise<T> {
    const parent = this.#context.getStore();

    if (parent) {
      const savepoint = `sp_${Date.now()}_${Math.floor(Math.random() * 1_000_000)}`;
      await parent.client.query(`SAVEPOINT ${savepoint}`);

      try {
        const result = await fn();
        await parent.client.query(`RELEASE SAVEPOINT ${savepoint}`);
        return result;
      } catch (error) {
        await parent.client.query(`ROLLBACK TO SAVEPOINT ${savepoint}`);
        await parent.client.query(`RELEASE SAVEPOINT ${savepoint}`);
        throw error;
      }
    }

    const client = await this.pool.connect();

    try {
      await client.query('BEGIN');
      const result = await this.#context.run({ client }, fn);
      await client.query('COMMIT');
      return result;
    } catch (error) {
      try {
        await client.query('ROLLBACK');
      } catch {
        // Preserve the original transaction error.
      }
      throw error;
    } finally {
      client.release();
    }
  }

  async close(): Promise<void> {
    if (this.#closed) return;
    this.#closed = true;
    await this.pool.end();
  }
}

/** Parse a JSON text column, falling back when it is empty or malformed. */
export function json<T>(value: unknown, fallback: T): T {
  if (typeof value !== 'string' || value === '') return fallback;

  try {
    return JSON.parse(value) as T;
  } catch {
    return fallback;
  }
}

-- Database-level guarantees behind "never mark an invoice paid unless payments cover it":
-- payments can't exceed the balance, can't change currency, and a total can't drop below what was paid.
CREATE TRIGGER payments_guard BEFORE INSERT ON payments
BEGIN
  SELECT RAISE(ABORT, 'PAYMENT_INVOICE_VOID')
    WHERE (SELECT state FROM invoices WHERE id = NEW.invoice_id) = 'void';
  SELECT RAISE(ABORT, 'PAYMENT_CURRENCY_MISMATCH')
    WHERE NEW.currency <> (SELECT currency FROM invoices WHERE id = NEW.invoice_id);
  SELECT RAISE(ABORT, 'PAYMENT_CLIENT_MISMATCH')
    WHERE NEW.client_id <> (SELECT client_id FROM invoices WHERE id = NEW.invoice_id);
  SELECT RAISE(ABORT, 'PAYMENT_EXCEEDS_BALANCE')
    WHERE NEW.amount_minor > (SELECT total_minor FROM invoices WHERE id = NEW.invoice_id)
                           - (SELECT COALESCE(SUM(amount_minor), 0) FROM payments WHERE invoice_id = NEW.invoice_id);
END;

CREATE TRIGGER payments_immutable BEFORE UPDATE OF invoice_id, client_id, amount_minor, currency ON payments
BEGIN
  SELECT RAISE(ABORT, 'PAYMENT_IMMUTABLE');
END;

CREATE TRIGGER invoices_total_covers_payments BEFORE UPDATE OF total_minor ON invoices
BEGIN
  SELECT RAISE(ABORT, 'INVOICE_TOTAL_BELOW_PAID')
    WHERE NEW.total_minor < (SELECT COALESCE(SUM(amount_minor), 0) FROM payments WHERE invoice_id = NEW.id);
END;

CREATE TRIGGER invoices_locked_when_paid BEFORE UPDATE OF currency, client_id, state ON invoices
WHEN EXISTS (SELECT 1 FROM payments WHERE invoice_id = NEW.id)
BEGIN
  SELECT RAISE(ABORT, 'INVOICE_HAS_PAYMENTS')
    WHERE NEW.currency <> OLD.currency OR NEW.client_id <> OLD.client_id OR NEW.state <> 'sent';
END;

CREATE VIEW invoice_balances AS
SELECT i.*,
       COALESCE(p.paid_minor, 0)                 AS paid_minor,
       i.total_minor - COALESCE(p.paid_minor, 0) AS balance_minor,
       COALESCE(p.payment_count, 0)              AS payment_count,
       p.last_paid_at                            AS last_paid_at
FROM invoices i
LEFT JOIN (
  SELECT invoice_id, SUM(amount_minor) AS paid_minor, COUNT(*) AS payment_count, MAX(paid_at) AS last_paid_at
  FROM payments GROUP BY invoice_id
) p ON p.invoice_id = i.id;

export function migrate(db: Db, now: () => string = () => new Date().toISOString()): number[] {
  db.exec(`CREATE TABLE IF NOT EXISTS schema_migrations (
    version    INTEGER PRIMARY KEY,
    name       TEXT NOT NULL,
    applied_at TEXT NOT NULL
  )`);
  const applied = new Set(db.all<{ version: number }>('SELECT version FROM schema_migrations').map((r) => r.version));
  const ran: number[] = [];
  for (const m of MIGRATIONS) {
    if (applied.has(m.version)) continue;
    db.tx(() => {
      db.exec(m.sql);
      db.run('INSERT INTO schema_migrations (version, name, applied_at) VALUES (:version, :name, :at)', {
        version: m.version,
        name: m.name,
        at: now(),
      });
    });
    ran.push(m.version);
  }
  return ran;
}
